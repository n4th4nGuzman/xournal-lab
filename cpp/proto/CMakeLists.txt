cmake_minimum_required(VERSION 3.20)
project(xl_proto LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(xl_proto main.cpp canvas.cpp)
target_link_libraries(xl_proto PRIVATE Qt6::Widgets)
if(MSVC)
  target_compile_options(xl_proto PRIVATE /W4)
else()
  target_compile_options(xl_proto PRIVATE -Wall -Wextra -O2)
endif()
