cmake_minimum_required(VERSION 3.20)
project(xournal_lab_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
find_package(ZLIB REQUIRED)
# Optional: PDF rasteriser (Qt6Pdf = pdfium; MSYS2: pacman -S mingw-w64-ucrt-x86_64-qt6-pdf) and printing. The build works without them (PDF pages stay blank).
find_package(Qt6 QUIET OPTIONAL_COMPONENTS Pdf PrintSupport)

# Logic and file format: no windows. Everything in core/ is built into one library; new files are picked up by a fresh `cmake` run.
file(GLOB CORE_SRC CONFIGURE_DEPENDS core/*.cpp)
add_library(xl_core STATIC ${CORE_SRC})
target_include_directories(xl_core PUBLIC core)
target_link_libraries(xl_core PUBLIC Qt6::Core Qt6::Gui ZLIB::ZLIB)
if(TARGET Qt6::Pdf)
  target_link_libraries(xl_core PUBLIC Qt6::Pdf)
  target_compile_definitions(xl_core PUBLIC XL_HAVE_PDF=1)
endif()
target_compile_options(xl_core PRIVATE -Wall -Wextra -O2)

# Command-line tools and tests: every tools/*.cpp and tests/*.cpp is its own executable (xl_<name>), run by `ctest`.
enable_testing()
file(GLOB TOOL_SRC CONFIGURE_DEPENDS tools/*.cpp)
foreach(src ${TOOL_SRC})
  get_filename_component(name ${src} NAME_WE)
  add_executable(xl_${name} ${src})
  target_link_libraries(xl_${name} PRIVATE xl_core)
  target_compile_options(xl_${name} PRIVATE -Wall -Wextra -O2)
endforeach()
file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/*.cpp)
foreach(src ${TEST_SRC})
  get_filename_component(name ${src} NAME_WE)
  add_executable(test_${name} ${src})
  target_link_libraries(test_${name} PRIVATE xl_core)
  target_compile_options(test_${name} PRIVATE -Wall -Wextra -O2)
  add_test(NAME ${name} COMMAND test_${name} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endforeach()

# The application (windows, tools, panels).
file(GLOB APP_SRC CONFIGURE_DEPENDS app/*.cpp app/*/*.cpp)
if(APP_SRC)
  set(APP_RC "")
  if(WIN32)   # the icon of the .exe (Explorer, taskbar); the window icon itself is set in main.cpp
    enable_language(RC)
    set(APP_RC app/xl_app.rc)
  endif()
  add_executable(xl_app WIN32 ${APP_SRC} app/resources.qrc ${APP_RC})
  target_include_directories(xl_app PRIVATE app)
  target_link_libraries(xl_app PRIVATE xl_core Qt6::Widgets)
  if(TARGET Qt6::PrintSupport)
    target_link_libraries(xl_app PRIVATE Qt6::PrintSupport)
    target_compile_definitions(xl_app PRIVATE XL_HAVE_PRINT=1)
  endif()
  target_compile_options(xl_app PRIVATE -Wall -Wextra -O2)
endif()
